\section{Preliminaries}\label{sec:preliminaries}

\subsection{Patterns, colored patterns and solutions}
\label{subsec:patterns}
\begin{definition}[Colored pattern on three vertices]
A \emph{colored pattern} on three vertices is a triple
\[
P=(A,N,C),
\]
where
\[
A,N\subseteq \binom{\{1,2,3\}}{2},
\qquad
A\cap N=\varnothing,
\qquad
C\subseteq\{1,2,3\}.
\]
Here, \(1,2,3\) denote the positions of an ordered triple
\[
v_1\prec v_2\prec v_3.
\]
The set \(A\) specifies the pairs that must be edges, \(N\) specifies the pairs that must be non-edges, and
\(C\) specifies the positions whose corresponding vertices are required to lie in the same part of the partition.
Pairs that do not belong to \(A\cup N\) are left unrestricted.

When \(C=\varnothing\), we call \(P\) an \emph{uncolored pattern}.


\end{definition}

\begin{definition}[Avoiding a colored pattern]\label{def:avoiding-colored-pattern}
Fix a graph \(G\), a solution
\[
S=(\prec,\mathcal Q)
\]
for \(G\), and a colored pattern
\[
P=(A,N,C).
\]

Let
\[
v_1\prec v_2\prec v_3
\]
be an ordered triple of \(G\). We say that the ordered triple \((v_1,v_2,v_3)\) \emph{avoids} \(P\) in \(G\) if at least one of the following
conditions holds:

\begin{description}
\item[\textup{(edge)}]\phantomsection\label{def:avoid:cond:edge}
There exists \(\{a,b\}\in A\) such that
\[
\{v_a,v_b\}\notin E(G).
\]

\item[\textup{(non-edge)}]\phantomsection\label{def:avoid:cond:nonedge}
There exists \(\{a,b\}\in N\) such that
\[
\{v_a,v_b\}\in E(G).
\]
\end{description}

\begin{description}
\item[\textup{(color)}]\phantomsection\label{def:avoid:cond:color}
There exist two distinct indices \(a,b\in C\) such that \(v_a\) and \(v_b\) lie in different parts of
\(\mathcal Q\).
\end{description}


The solution \(S\) \emph{avoids} \(P\) in \(G\) if every ordered triple of \(G\) avoids \(P\) in \(G\).

For a family of colored patterns \(\mathcal P\), the solution \(S\) avoids \(\mathcal P\) in \(G\) if it avoids every
colored pattern \(P\in\mathcal P\).


In all the colored patterns studied in this work, \(N=\varnothing\). Therefore, \nonedgecondition{} never holds, and an ordered triple fails to avoid a pattern precisely when it satisfies neither \edgecondition{} nor \colorcondition{}.

\end{definition}



\label{subsec:solutions-avoidance}
\begin{definition}[Solution and value]
Let \(G\) be a graph. A \emph{solution} for $G$ is a pair \(S=(\prec,\mathcal Q)\), where \(\prec\) is a total order on \(V(G)\)
and \(\mathcal Q=\{V_1,\dots,V_k\}\) is a partition of \(V(G)\) into \(k\) parts.
The \emph{value} of \(S\) is \(|\mathcal Q|=k\).
\end{definition}

\begin{definition}[\(P\)-coloring and \(P\)-chromatic number]

such that
\[

\subsection{Notation for colored versions and induced parameters}
\label{subsec:colored-notation}

\begin{definition}[Colored versions and families associated with a pattern]
\label{def:colored-versions-and-families}
Let
\[
P=(A,N,\varnothing)
\]
be an uncolored pattern.

For every set
\[
I\subseteq\{1,2,3\},
\]
we define the colored version
\[
P_I:=(A,N,I).
\]

In subscripts, we omit braces and commas. Thus, for example,
\[
P_{12}:=P_{\{1,2\}}=(A,N,\{1,2\}),
\]
\[
P_{13}:=P_{\{1,3\}}=(A,N,\{1,3\}),
\]
and
\[
P_{123}:=P_{\{1,2,3\}}=(A,N,\{1,2,3\}).
\]

Given distinct sets
\[
I_1,\dots,I_\ell\subseteq\{1,2,3\},
\qquad \ell\ge 2,
\]
we use the hyphenated notation
\[
P_{I_1\text{-}\cdots\text{-}I_\ell}
:=
\{P_{I_1},\dots,P_{I_\ell}\}
\]
for the family formed by the corresponding colored versions of \(P\).

For example,
\[
P_{12\text{-}13}
=
\{P_{12},P_{13}\},
\]
\[
P_{12\text{-}23}
=
\{P_{12},P_{23}\},
\]
and
\[
P_{12\text{-}13\text{-}23}
=
\{P_{12},P_{13},P_{23}\}.
\]
\end{definition}




\begin{definition}[Notation for the induced parameters]
\label{def:notation-induced-parameters}
Let
\[
P=(A,N,\varnothing)
\]
be an uncolored pattern, and let
\[
I_1,\dots,I_\ell\subseteq\{1,2,3\}
\]
be distinct sets satisfying
\[
|I_j|\ge 2
\qquad
\text{for every }j=1,\dots,\ell.

\[
P_{I_1\text{-}\cdots\text{-}I_\ell}(G)

\minClases_{\{P_{I_1},\dots,P_{I_\ell}\}}(G)

\subsection{Decision problems}
\label{subsec:decision-problems}


Let \(\mathcal P\) be a family of colored patterns. We consider the following three decision problems.
 
\begin{problem}[\(\mathcal P\)-Avoidance]
\textbf{Input:} A graph \(G\) and an integer \(k\).

\textbf{Question:} Does there exist a solution
\[
S=(\prec,\mathcal Q),
\qquad
\mathcal Q=\{V_1,\dots,V_k\},
\]
for \(G\) such that \(S\) avoids every colored pattern in \(\mathcal P\)?
\end{problem}

\begin{problem}[\(\mathcal P\)-Avoidance with a given order]
\textbf{Input:} A graph \(G\), a total order \(\prec\) of \(V(G)\), and an integer \(k\).

\textbf{Question:} Does there exist a partition
\[
\mathcal Q=\{V_1,\dots,V_k\},
\]
of \(V(G)\)
such that the solution
\[
S=(\prec,\mathcal Q)
\]
for \(G\) avoids every colored pattern in \(\mathcal P\)?
\end{problem}



\begin{problem}[\(\mathcal P\)-Avoidance with a given partition]
\textbf{Input:} A graph \(G\) and a partition
\[
\mathcal Q=\{V_1,\dots,V_k\}
\]
of \(V(G)\).

\textbf{Question:} Does there exist a total order \(\prec\) of \(V(G)\) such that the solution
\[
S=(\prec,\mathcal Q)
\]
for \(G\) avoids every colored pattern in \(\mathcal P\)?
\end{problem}

When \(\mathcal P=\{P\}\) consists of a single colored pattern, we simply write \textsc{\(P\)-Avoidance} instead of
\textsc{\(\{P\}\)-Avoidance}.


\begin{observation}[Membership in NP]
\label{obs:avoidance-in-NP}
Let \(\mathcal P\) be a fixed finite family of patterns. Then the following
problems belong to \(\mathrm{NP}\):
\begin{itemize}
    \item \(\mathcal P\)-Avoidance;
    \item \(\mathcal P\)-Avoidance with a given order;
    \item \(\mathcal P\)-Avoidance with a given partition.
\end{itemize}
\end{observation}

\begin{proof}
For \(\mathcal P\)-Avoidance, a certificate consists of a total order
\(\prec\) of \(V(G)\) and a partition \(\mathcal Q\) of \(V(G)\).
For the variant with a given order, only the partition \(\mathcal Q\)
needs to be provided, while for the variant with a given partition, only
the order \(\prec\) needs to be provided.

In each case, the certificate can be verified in polynomial time by
examining every ordered triple
\[
v_1\prec v_2\prec v_3
\]
and every pattern \(P\in\mathcal P\), and checking whether at least one of
the \edgecondition{}, \nonedgecondition{}, or \colorcondition{} holds.

Therefore, all three decision problems belong to \(\mathrm{NP}\).
\end{proof}

Consequently, throughout the rest of the paper, whenever one of these
problems is shown to be NP-hard, it follows immediately that the problem
is NP-complete.



\subsection{General structural results}
\label{subsec:general-structural-results}




\begin{corollary}[Value one of the avoidance parameter]
\label{cor:minClasses-value-one}
Let \(P=(A,N,C)\) be a pattern with \(|C|\ge2\). Then, for every graph \(G\),
\[
\minClases_P(G)=1
\quad\Longleftrightarrow\quad
\chi_P(G)=1
\quad\Longleftrightarrow\quad
G\in\mathcal C_P.
\]
\end{corollary}



\begin{lemma}[Monotonicity under inclusion of pattern families]
\label{lem:monotone-patterns}
Let \(\mathcal P\) and \(\mathcal P'\) be two finite families of patterns.
If
\[
\mathcal P\subseteq\mathcal P',
\]
then, for every graph \(G\),
\[
\minClases_{\mathcal P}(G)
\le
\minClases_{\mathcal P'}(G).
\]
\end{lemma}

\begin{proof}
Every solution that avoids all the patterns in \(\mathcal P'\) also avoids all the patterns in
\(\mathcal P\), since
\[
\mathcal P\subseteq\mathcal P'.
\]
Therefore, the set of feasible solutions for
\(\minClases_{\mathcal P'}(G)\) is contained in the set of feasible solutions for
\(\minClases_{\mathcal P}(G)\). Consequently,
\[
\minClases_{\mathcal P}(G)
\le
\minClases_{\mathcal P'}(G).
\]
\end{proof}

\begin{lemma}[Monotonicidad por inclusión de patrones]\label{lem:monotone-patterns}
Sea \(G\) un grafo y sean \(\mathcal P,\mathcal P'\) dos familias de patrones coloreados.
Si \(\mathcal P\subseteq \mathcal P'\), entonces
\[
\minClases_{\mathcal P}(G)\;\le\;\minClases_{\mathcal P'}(G).
\]
\end{lemma}

\begin{proof}
Toda solución que evita simultáneamente todos los patrones de \(\mathcal P'\) evita, en particular,
todos los patrones de \(\mathcal P\) (pues \(\mathcal P\subseteq \mathcal P'\)).
Por lo tanto, el conjunto de soluciones factibles para \(\minClases_{\mathcal P'}(G)\) es un subconjunto
del conjunto de soluciones factibles para \(\minClases_{\mathcal P}(G)\), y el mínimo número de clases
no puede disminuir al imponer más restricciones. En consecuencia,
\(\minClases_{\mathcal P}(G)\le \minClases_{\mathcal P'}(G)\).
\end{proof}


\begin{definition}[Order on patterns]
\label{def:pat-order}
Let
\[
P=(A,N,C)
\qquad\text{and}\qquad
P'=(A',N',C')
\]
be two patterns. We write
\[
P\preceq P'
\]
if
\[
A\subseteq A',
\qquad
N\subseteq N',
\qquad
C\subseteq C'.
\]
\end{definition}



\begin{lemma}[Monotonicity with respect to \(\preceq\)]
\label{lem:pat-monotone}
Let \(P\) and \(P'\) be two patterns such that
\[
P\preceq P'.
\]
Then the following statements hold.

\begin{enumerate}
\item Every solution that avoids \(P\) also avoids \(P'\).

\item For every graph \(G\),
\[
\minClases_{P'}(G)
\le
\minClases_P(G).
\]

\item The associated graph classes satisfy
\[
\mathcal C_P\subseteq\mathcal C_{P'},
\]
and consequently, for every graph \(G\),
\[
\chi_{P'}(G)
\le
\chi_P(G).
\]
\end{enumerate}
\end{lemma}

\begin{proof}
Let
\[
P=(A,N,C)
\qquad\text{and}\qquad
P'=(A',N',C'),
\]
with
\[
A\subseteq A',
\qquad
N\subseteq N',
\qquad
C\subseteq C'.
\]

Let \(S=(\prec,\mathcal Q)\) be a solution that avoids \(P\), and consider an ordered triple
\[
v_1\prec v_2\prec v_3.
\]
Since the triple avoids \(P\), at least one of the following conditions holds.

\begin{itemize}
\item Suppose that the triple avoids \(P\) by the \edgecondition. Then there exists
\[
\{a,b\}\in A
\]
such that
\[
\{v_a,v_b\}\notin E(G).
\]
Since \(A\subseteq A'\), the same pair belongs to \(A'\), and therefore the triple also avoids \(P'\) by the
\edgecondition.

\item Suppose that the triple avoids \(P\) by the \nonedgecondition. Then there exists
\[
\{a,b\}\in N
\]
such that
\[
\{v_a,v_b\}\in E(G).
\]
Since \(N\subseteq N'\), the same pair belongs to \(N'\), and therefore the triple also avoids \(P'\) by the
\nonedgecondition.

\item Suppose that the triple avoids \(P\) by the \colorcondition. Then there exist two distinct indices
\[
a,b\in C
\]
such that \(v_a\) and \(v_b\) lie in different parts of \(\mathcal Q\). Since \(C\subseteq C'\), we also have
\(a,b\in C'\). Hence the triple avoids \(P'\) by the \colorcondition.
\end{itemize}

Thus every ordered triple avoids \(P'\), and therefore \(S\) avoids \(P'\). This proves the first statement.

It follows immediately that every feasible solution for \(\minClases_P(G)\) is also feasible for
\(\minClases_{P'}(G)\). Hence
\[
\minClases_{P'}(G)
\le
\minClases_P(G).
\]

Finally, let \(H\in\mathcal C_P\). By definition, there exists an order \(\prec_H\) such that the one-part solution
\[
\bigl(\prec_H,\{V(H)\}\bigr)
\]
avoids \(P\) in \(H\). By the first statement, this same solution avoids \(P'\), and therefore
\[
H\in\mathcal C_{P'}.
\]
Thus
\[
\mathcal C_P\subseteq\mathcal C_{P'}.
\]

Now let
\[
V(G)=V_1\,\dot\cup\,\cdots\,\dot\cup\,V_k
\]
be a \(P\)-coloring of \(G\). Then
\[
G[V_i]\in\mathcal C_P\subseteq\mathcal C_{P'}
\qquad
\text{for every }i=1,\dots,k.
\]
Therefore the same partition is also a \(P'\)-coloring of \(G\), and consequently
\[
\chi_{P'}(G)
\le
\chi_P(G).
\]
\end{proof}






\begin{lemma}[Valid solutions induce \(P\)-colorings]
\label{lem:solution-induces-P-coloring}
Let \(P\) be a pattern, let \(G\) be a graph, and let
\[
S=(\prec,\mathcal Q)
\]
be a solution that avoids \(P\) in \(G\). Then \(\mathcal Q\) is a
\(P\)-coloring of \(G\).
\end{lemma}

\begin{proof}
Let
\[

Therefore the one-part solution on \(G[V_i]\) avoids \(P\), and thus
\[
G[V_i]\in\mathcal C_P.
\]
Hence \(\mathcal Q\) is a \(P\)-coloring of \(G\).
\end{proof}

\subsection{The given-order problem and constraint graphs}
\label{subsec:constraint-graphs}

\paragraph{Intuition.}
Once the order \(\prec\) is fixed, every occurrence of a pattern
\[
P=(A,N,\{a,b\})
\]
imposes a single constraint: the vertices occupying positions \(a\) and \(b\)
must lie in different parts. We represent this constraint by adding an edge
between those two vertices in an auxiliary graph. Hence, finding a valid
partition of \(G\) is equivalent to properly coloring this auxiliary graph.

\begin{definition}[Constraint graph]
\label{def:constraint-graph}
Let \(\mathcal P\) be a finite family of patterns such that
\[
|C|=2
\qquad
\text{for every }(A,N,C)\in\mathcal P.
\]
Let \(G\) be a graph, and let \(\prec\) be a total order of \(V(G)\).

The \emph{constraint graph} associated with \(G\), \(\prec\), and
\(\mathcal P\), denoted by
\[
R_{\mathcal P}(G,\prec),
\]
is defined as follows. Its vertex set is
\[
V\bigl(R_{\mathcal P}(G,\prec)\bigr):=V(G).
\]

For every ordered triple
\[
v_1\prec v_2\prec v_3
\]
and every pattern
\[
P'=(A',N',\{a,b\})\in\mathcal P,
\]
if neither the \edgecondition{} nor the \nonedgecondition{} holds for \(P'\),
we add the edge
\[
\{v_a,v_b\}
\]

to \(R_{\mathcal P}(G,\prec)\).
In other words, whenever an ordered triple satisfies all the edge and
non-edge requirements of a pattern \(P'\), the constraint graph records that
the two vertices indicated by the color set of \(P'\) must lie in different
parts.
\end{definition}
\begin{lemma}[Constraint graph characterization]
\label{lem:constraint-graph-characterization}
Let \(\mathcal P\) be a finite family of patterns such that
\[
|C|=2
\qquad
\text{for every }(A,N,C)\in\mathcal P.
\]
Let \(G\) be a graph, let \(\prec\) be a total order of \(V(G)\), and let
\[
R:=R_{\mathcal P}(G,\prec)
\]
be the corresponding constraint graph.

Then, for every integer \(k\ge 1\), the following statements are equivalent:

\begin{enumerate}
\item There exists a partition \(\mathcal Q\) of \(V(G)\) into at most
\(k\) parts such that
\[
(\prec,\mathcal Q)
\]
avoids every pattern in \(\mathcal P\) in \(G\).

\item
\[
\chi(R)\le k.
\]
\end{enumerate}


\end{lemma}

\begin{proof}
Suppose first that there exists a partition
\[
\mathcal Q=\{V_1,\dots,V_q\},
\qquad
q\le k,
\]
such that
\[
(\prec,\mathcal Q)
\]
avoids every pattern in \(\mathcal P\) in \(G\).

We show that \(\mathcal Q\) is a proper coloring of \(R\).

Let
\(
\{x,y\}\in E(R).
\)
By the definition of the constraint graph, there exist an ordered triple
\[
v_1\prec v_2\prec v_3
\]
and a pattern
\[
P'=(A',N',\{a,b\})\in\mathcal P
\]
such that
\[
x=v_a,
\qquad
y=v_b,
\]
and neither the \edgecondition{} nor the \nonedgecondition{} holds for
\(P'\).

Since the solution \((\prec,\mathcal Q)\) avoids \(P'\), the
\colorcondition{} must hold. Therefore, \(v_a\) and \(v_b\) lie in
different parts of \(\mathcal Q\). Hence \(x\) and \(y\) receive different
colors.

Thus every edge of \(R\) has endpoints with different colors, so \(R\)
admits a proper coloring with \(q\le k\) colors. Consequently,
\[
\chi(R)\le k.
\]

Conversely, suppose that
\[
\chi(R)\le k.
\]
Let \(c\) be a proper coloring of \(R\) with at most \(k\) colors.
Since
\[
V(R)=V(G),
\]
we take the color classes of \(c\) as a partition \(\mathcal Q\) of
\(V(G)\) into at most \(k\) parts.

We claim that
\[
(\prec,\mathcal Q)
\]
avoids every pattern in \(\mathcal P\) in \(G\).

Let
\[
v_1\prec v_2\prec v_3
\]
be an ordered triple, and let
\[
P'=(A',N',\{a,b\})\in\mathcal P.
\]

If the \edgecondition{} or the \nonedgecondition{} holds for \(P'\), then
the triple avoids \(P'\).

Suppose instead that neither the \edgecondition{} nor the
\nonedgecondition{} holds. By the definition of the constraint graph,
\[
\{v_a,v_b\}\in E(R).
\]
Since \(c\) is a proper coloring of \(R\), the vertices \(v_a\) and \(v_b\)
receive different colors, and therefore lie in different parts of
\(\mathcal Q\). Hence the \colorcondition{} holds, so the triple avoids
\(P'\).

Therefore every ordered triple avoids every pattern in \(\mathcal P\), and
thus
\[
(\prec,\mathcal Q)
\]
avoids \(\mathcal P\) in \(G\).
\end{proof}






\begin{corollary}[Constraint graph characterization for two parts]
\label{cor:constraint-graph-two-parts}
Under the assumptions of
Lemma~\ref{lem:constraint-graph-characterization}, there exists a partition
\(\mathcal Q\) of \(V(G)\) into at most two parts such that
\[
(\prec,\mathcal Q)
\]
avoids \(\mathcal P\) in \(G\) if and only if
\[
R_{\mathcal P}(G,\prec)
\]
is bipartite.
\end{corollary}

\begin{proof}
By Lemma~\ref{lem:constraint-graph-characterization}, such a partition
exists if and only if the constraint graph is \(2\)-colorable. A graph is
\(2\)-colorable if and only if it is bipartite.
\end{proof}

\begin{theorem}[Polynomial-time algorithm for two parts]
\label{thm:given-order-two-parts-polynomial}
Let \(\mathcal P\) be a fixed finite family of patterns such that
\[
|C|=2
\qquad
\text{for every }(A,N,C)\in\mathcal P.
\]
Then the \(\mathcal P\)-Avoidance problem with a given order, restricted
to instances with
\(
k=2,
\)
can be solved in time
\(
O\bigl(|V(G)|^3\bigr).
\)
\end{theorem}

\begin{proof}
Algorithm~\ref{alg:two-parts-given-order} constructs precisely the
constraint graph
\(
R_{\mathcal P}(G,\prec)
\)
from Definition~\ref{def:constraint-graph}. By
Corollary~\ref{cor:constraint-graph-two-parts}, the algorithm returns
\textbf{YES} if and only if the given ordered graph admits a valid partition
into at most two parts.

It remains to analyze its running time. Let
\(
n:=|V(G)|.
\)
There are
\(
O(n^3)
\)
ordered triples
\(
v_1\prec v_2\prec v_3.
\)
For each triple and each pattern in \(\mathcal P\), testing the required
edges and non-edges takes constant time when \(G\) is represented by an
adjacency matrix. Since \(\mathcal P\) is fixed, the construction of \(R\)
takes
\(
O(n^3)
\)
time.
The graph \(R\) has \(n\) vertices and at most \(O(n^2)\) edges.
Bipartiteness can be tested by BFS or DFS in
\[
O\bigl(|V(R)|+|E(R)|\bigr)
\subseteq
O(n^2)
\]
time. Therefore, the total running time is
\(
O(n^3).
\)
\end{proof}



\begin{remark}[Relation with thinness and proper thinness]
The constraint graph \(R_{\mathcal P}(G,\prec)\) is analogous to the auxiliary
graphs used for thinness and proper thinness.

For a fixed order \(\prec\), the graph \(G_{\prec}\) associated with thinness
has an edge between two vertices whenever they cannot belong to the same part
of a partition consistent with \(\prec\). Similarly, the graph
\(\widetilde{G}_{\prec}\) associated with proper thinness records pairs of
vertices that cannot belong to the same part of a strongly consistent
partition~\cite{BonomoDeEstrada2019}.

In both cases, the chromatic number of the auxiliary graph equals the minimum
number of parts compatible with the fixed order. Thus,
\(R_{\mathcal P}(G,\prec)\) can be viewed as an extension of the same idea to
families of forbidden colored patterns with color sets of size two.
\end{remark}




\begin{algorithm}[ht]
\caption{\textsc{Two parts given order}}
\label{alg:two-parts-given-order}
\begin{algorithmic}[1]
\Require A graph \(G\), a total order \(\prec\) of \(V(G)\), and a finite
family \(\mathcal P\) of patterns such that every pattern has a color set
of size \(2\).
\Ensure \textbf{YES} if there exists a partition \(\mathcal Q\) of \(V(G)\)
into at most two parts such that \((\prec,\mathcal Q)\) avoids
\(\mathcal P\) in \(G\); otherwise \textbf{NO}.

\State Construct a graph \(R\) with
\[
V(R)\gets V(G),
\qquad
E(R)\gets\varnothing.
\]

\ForAll{ordered triples \(v_1\prec v_2\prec v_3\)}
    \ForAll{\(P'=(A',N',\{a,b\})\in\mathcal P\)}
        \If{neither the \edgecondition{} nor the \nonedgecondition{}
        holds for \(P'\) on \((v_1,v_2,v_3)\)}
            \State \(E(R)\gets E(R)\cup\{\{v_a,v_b\}\}\)
        \EndIf
    \EndFor
\EndFor
\If{\(R\) is bipartite}
    \State \Return \textbf{YES}
\Else
    \State \Return \textbf{NO}
\EndIf
\end{algorithmic}
\end{algorithm}

\begin{theorem}\label{thm:seg1-bip}
Sea \(G=(V,E)\) un grafo. Entonces \(G\) es bipartito si y solo si \(\segone(G)=1\).
\end{theorem}


\begin{proof}
\noindent(\(\Rightarrow\)) Supongamos que \(G\) es bipartito. Entonces existe una partición
\(V=V_1\,\dot\cup\,V_2\) tal que \(V_1\) y \(V_2\) son independientes.
Consideremos la partición en una sola clase \(\mathcal{Q}:=\{V\}\) y se fija un orden \(\prec\) \emph{por bloques}, en el sentido de que
\[
\text{para todo } v_1\in V_1 \text{ y todo } v_2\in V_2,\qquad v_1\prec v_2.
\]

Dentro de cada bloque \(V_1\) y \(V_2\) se toma un orden arbitrario.

Sea \(P_1=(A,N,C)\) el patrón definido previamente para \(\segone\), con
\(A=\{\{1,2\},\{2,3\}\}\), \(N=\varnothing\) y \(C=\{\{1,2\}\}\).
Verifiquemos que \((\prec,\mathcal{Q})\) evita \(P_1\),
sea \(v_1\prec v_2\prec v_3\) una terna de vértices.


Supongamos que simultáneamente \(v_1\sim v_2\) y \(v_2\sim v_3\), en particular \(v_1\in V_1\) y \(v_2\in V_2\),(pues \(V_1\) es independiente y el orden ubica a todos los vértices de \(V_2\) a la derecha de los de \(V_1\)),
además como \(v_2\prec v_3\) se deduce \(v_3\in V_2\), pero esto es absurdo porque \(V_2\) es independiente, por lo tanto,
se tiene \(v_2\not\sim v_3\).

Luego \(\segone(G)\le 1\) y, como siempre vale \(\segone(G)\ge 1\), se concluye \(\segone(G)=1\).




\medskip
\noindent(\(\Leftarrow\)) Supongamos ahora que \(\segone(G)=1\),
por definición, existe una solución \((\prec,\mathcal{Q})\) con \(|\mathcal{Q}|=1\) que evita \(P_1\),
en particular, \(\mathcal{Q}=\{V\}\), de modo que para toda tripla de vértices
\(r\prec s\prec t\) no pueden ocurrir simultáneamente
\(r\sim s\) y \(s\sim t\) o sea,
\[
(r\sim s)\ \Rightarrow\ (s\not\sim t)
\qquad\text{para todo } r\prec s\prec t
\]
En particular, si un vértice tiene algún vecino a la izquierda, entonces no puede tener vecinos a la derecha 
y viceversa.


Se definen los conjuntos
\[
L:=\{\,s\in V:\ \text{no existe } t\in V \text{ con } s\prec t \text{ y } s\sim t\,\},
\]
\[
D:=\{\,s\in V:\ \text{no existe } r\in V \text{ con } r\prec s \text{ y } r\sim s\,\}.
\]
Es decir, \(L\) es el conjunto de vértices sin vecinos a la derecha, y \(D\) el de vértices sin vecinos a la izquierda.

Veamos que \(L\) es independiente, si \(r\prec s\) con \(r,s\in L\) y \(r\sim s\), entonces \(r\) tendría
un vecino a la derecha (\(s\)), contradicción con la definición de \(L\),
luego no hay aristas internas en \(L\), analogo para \(D\).



Por la observación anterior, todo vértice no aislado pertenece a \(L\) o a \(D\) y
los vértices aislados no afectan la independencia, y pueden asignarse al final a cualquiera de los dos conjuntos.
Luego obtuvimos una partición de \(V\) en dos conjuntos independientes, y por lo tanto \(G\) es bipartito.
\end{proof}

\begin{theorem}\label{thm:bip12-chi}
Let \(G\) be a graph with \(\chi(G)\ge 2\). Then
\[
\bipparam{12}(G)=\chi(G)-1.
\]
\end{theorem}

\begin{proof}
We prove both inequalities.

\medskip
\noindent\textbf{First, \(\chi(G)-1\le \bipparam{12}(G)\).}

Let
\[
k:=\bipparam{12}(G),
\]
and let
\[
S=(\prec,\mathcal Q),
\qquad
\mathcal Q=\{V_1,\dots,V_k\},
\]
be a solution of value \(k\) that avoids \(\operatorname{bip}_{12}\).
We construct a proper coloring of \(G\) with at most \(k+1\) colors. For each vertex \(v\in V_i\), define
\[
c(v)=
\begin{cases}
0,
& \text{if \(v\) has an earlier neighbor in \(V_i\)},\\[1mm]
i,
& \text{otherwise}.
\end{cases}
\]

We first observe that if \(c(v)=0\), then \(v\) has no later neighbor. Indeed, there exists
\(u\in V_i\) such that
\[
u\prec v
\qquad\text{and}\qquad
u\sim v.
\]
If \(v\) had a later neighbor \(w\), then
\[
u\prec v\prec w,
\qquad
u\sim v,
\qquad
v\sim w.
\]
Moreover, \(u\) and \(v\) lie in the same part of \(\mathcal Q\). Therefore, the ordered triple
\((u,v,w)\) satisfies neither \edgecondition{} nor \colorcondition{} for
\(\operatorname{bip}_{12}\), contradicting that \(S\) avoids the pattern.

We now show that \(c\) is a proper coloring.

Suppose first that
\[
x\prec y
\qquad\text{and}\qquad
c(x)=c(y)=0.
\]
If \(x\sim y\), then \(x\) would have a later neighbor, contradicting the previous observation. Hence
\(x\not\sim y\).

Now suppose that
\[
x\prec y
\qquad\text{and}\qquad
c(x)=c(y)=i
\]
for some \(i\in\{1,\dots,k\}\). Then \(x,y\in V_i\). If \(x\sim y\), the vertex \(y\) would have an earlier
neighbor in \(V_i\), and therefore \(c(y)=0\), a contradiction.

Thus every color class of \(c\) is independent. Hence
\[
\chi(G)\le k+1,
\]
and therefore
\[
\chi(G)-1\le k=\bipparam{12}(G).
\]

\medskip
\noindent\textbf{Now, \(\bipparam{12}(G)\le \chi(G)-1\).}

Let
\[
k:=\chi(G),
\]
and fix a proper \(k\)-coloring of \(G\), represented by a partition
\[
V(G)=C_1\,\dot\cup\,C_2\,\dot\cup\,\cdots\,\dot\cup\,C_k,
\]
where each \(C_i\) is independent.

Define a partition
\[
\mathcal Q=\{V_1,\dots,V_{k-1}\}
\]
by
\[
V_1:=C_1\cup C_k,
\qquad
V_i:=C_i
\quad\text{for }i=2,\dots,k-1.
\]
Define the order \(\prec\) by color blocks:
\[
C_1\prec C_2\prec\cdots\prec C_k,
\]
taking an arbitrary order inside each \(C_i\).

We claim that
\[
S=(\prec,\mathcal Q)
\]
avoids \(\operatorname{bip}_{12}\).

Let
\[
v_1\prec v_2\prec v_3
\]
be an ordered triple. If
\[
v_1\not\sim v_2
\qquad\text{or}\qquad
v_2\not\sim v_3,
\]
then the triple avoids \(\operatorname{bip}_{12}\) by \edgecondition.

Suppose then that
\[
v_1\sim v_2
\qquad\text{and}\qquad
v_2\sim v_3.
\]
We show that \(v_1\) and \(v_2\) lie in different parts of \(\mathcal Q\).

Suppose, towards a contradiction, that \(v_1\) and \(v_2\) lie in the same part. Since
\[
V_i=C_i
\qquad\text{for }i=2,\dots,k-1,
\]
and every \(C_i\) is independent, necessarily
\[
v_1,v_2\in V_1=C_1\cup C_k.
\]
Because \(v_1\sim v_2\), they belong to different color classes. Since
\[
C_1\prec C_k
\qquad\text{and}\qquad
v_1\prec v_2,
\]
we obtain
\[
v_1\in C_1
\qquad\text{and}\qquad
v_2\in C_k.
\]

The class \(C_k\) is the last block of the order. Hence \(v_2\prec v_3\) implies
\[
v_3\in C_k.
\]
Since \(C_k\) is independent, this gives
\[
v_2\not\sim v_3,
\]
a contradiction.

Therefore \(v_1\) and \(v_2\) lie in different parts of \(\mathcal Q\), and the triple avoids
\(\operatorname{bip}_{12}\) by \colorcondition. Thus \(S\) avoids \(\operatorname{bip}_{12}\), and
\[
\bipparam{12}(G)\le k-1=\chi(G)-1.
\]

Combining both inequalities, we conclude that
\[
\bipparam{12}(G)=\chi(G)-1.
\]
\end{proof}

\begin{theorem}[
\texorpdfstring{\(\chi_P(G)=\left\lceil\chi(G)/2\right\rceil\) for \(P=\bipname\)}
{P-chromatic number of the bipartite pattern}
]
\label{thm:bip-P-chromatic-number}
Let
\[
P=\bipname
\]
be the uncolored bipartite pattern. Then, for every graph \(G\),
\[
\chi_P(G)
=
\left\lceil\frac{\chi(G)}{2}\right\rceil.
\]
\end{theorem}

\begin{proof}
Recall that \(\mathcal C_P\) is the class of bipartite graphs. We prove both
inequalities.

First, let
\[
k:=\chi(G),
\]
and fix a proper \(k\)-coloring of \(G\), represented by a partition
\[
V(G)=C_1\,\dot\cup\,\cdots\,\dot\cup\,C_k,
\]
where each \(C_i\) is independent.

Let
\[
q:=\left\lceil\frac{k}{2}\right\rceil.
\]
We define a partition
\[
V(G)=V_1\,\dot\cup\,\cdots\,\dot\cup\,V_q
\]
by joining the color classes in pairs:
\[
V_i:=C_{2i-1}\cup C_{2i}
\qquad
\text{for }i=1,\dots,\left\lfloor\frac{k}{2}\right\rfloor.
\]
If \(k\) is odd, we additionally set
\[
V_q:=C_k.
\]

Each graph \(G[V_i]\) is bipartite: when \(V_i=C_{2i-1}\cup C_{2i}\), the
sets \(C_{2i-1}\) and \(C_{2i}\) form a bipartition of \(G[V_i]\); if
\(V_q=C_k\), then \(G[V_q]\) is independent and hence bipartite. Therefore,
\[
G[V_i]\in\mathcal C_P
\qquad
\text{for every }i=1,\dots,q.
\]
Thus the partition is a \(P\)-coloring of \(G\), and
\[
\chi_P(G)
\le
q
=
\left\lceil\frac{\chi(G)}{2}\right\rceil.
\]

Conversely, let
\[
q:=\chi_P(G),
\]
and let
\[
V(G)=V_1\,\dot\cup\,\cdots\,\dot\cup\,V_q
\]
be a \(P\)-coloring of \(G\). Then
\[
G[V_i]\in\mathcal C_P
\qquad
\text{for every }i=1,\dots,q,
\]
so every \(G[V_i]\) is bipartite.

For each \(i\), choose a bipartition
\[
V_i=L_i\,\dot\cup\,R_i,
\]
where \(L_i\) and \(R_i\) are independent sets. Consequently, the sets
\[
L_1,R_1,\dots,L_q,R_q
\]
form a proper coloring of \(G\) with at most \(2q\) colors. Hence
\[
\chi(G)\le 2q.
\]
Therefore,
\[
q\ge
\left\lceil\frac{\chi(G)}{2}\right\rceil,
\]
and thus
\[
\chi_P(G)
\ge
\left\lceil\frac{\chi(G)}{2}\right\rceil.
\]

Combining both inequalities, we conclude that
\[
\chi_P(G)
=
\left\lceil\frac{\chi(G)}{2}\right\rceil.
\]
\end{proof}

\paragraph{Intuition.}
The following reduction is designed by first considering the reverse implication.
For each edge \(xy\in E(X)\) as always, we ask: "How can we force the endpoints of every edge of a graph \(X\) to lie in different parts", we add a vertex \(w_{xy}\) so that \(x,y,w_{xy}\) form a triangle.

\(\operatorname{bip}_{12}\), \(\operatorname{bip}_{23}\), and \(\operatorname{bip}_{13}\),
the three vertices of this triangle must lie in pairwise different parts, regardless of their relative order.\eric{Considerar poner estas cosas de intuitions en un lemma u observation.}

\begin{Claim}\label{cl:seg123-le-chi}
Para todo grafo \(G\), se cumple
\[
\segonetwothree(G)\le \chi(G).
\]
\end{Claim}

\begin{proof}
Sea \(k=\chi(G)\), consideremos una partición de \(V(G)\) en \(k\) conjuntos independientes,
\[
V \;=\; C_1 \,\dot\cup\, C_2 \,\dot\cup\, \cdots \,\dot\cup\, C_k.
\]
Se considera la solución \(S=(\prec,\mathcal{Q})\) definida por
\[
\mathcal{Q}:=\{V_1,\dots,V_k\}\qquad\text{donde}\qquad V_i:=C_i \ \ \text{para todo } i\in\{1,\dots,k\},
\]
y se fija un orden \(\prec\) por bloques imponiendo
\[
C_1 \prec C_2 \prec \cdots \prec C_k,
\]
con un orden arbitrario dentro de cada bloque.

Veamos que \(S\) evita simultáneamente los patrones \(P_1,P_2,P_3\),
como cada \(V_i=C_i\) es independiente, para toda terna \(r\prec s\prec t\) se tiene:
si \(r\) y \(s\) pertenecen a una misma clase, entonces \(r\not\sim s\); y si \(s\) y \(t\) pertenecen
a una misma clase, entonces \(s\not\sim t\); y si \(r\) y \(t\) pertenecen a una misma clase, entonces
\(r\not\sim t\).
En particular, en ninguna terna pueden realizarse simultáneamente las aristas exigidas por el conjunto
\(A\) del patrón junto con la igualdad de clase exigida por \(C\). Por lo tanto, toda terna evita
\(P_1\), evita \(P_2\) y evita \(P_3\), luego \(S\) evita \(\{P_1,P_2,P_3\}\).

Con esto, \(G\) admite una solución que evita \(P_1,P_2,P_3\) con \(k\) clases, luego
\[
\segonetwothree(G)\le k=\chi(G).
\]
\end{proof}

\subsubsection{The Mycielski counterexample}



\begin{definition}[Paso de Mycielski]
Sea \(M_j=(V_j,E_j)\) un grafo, con \(V_j=\{v_0,\dots,v_{n-1}\}\).
Se construye \(M_{j+1}\) introduciendo una copia
\[
W_j:=\{w_0,\dots,w_{n-1}\},
\]
y un vértice adicional \(w\). Se fija
\[
V(M_{j+1}) \;:=\; V_j \,\dot\cup\, W_j \,\dot\cup\, \{w\}.
\]
La adyacencia en \(M_{j+1}\) se determina imponiendo, para todo \(i,\ell\in\{0,\dots,n-1\}\),
\[
v_i \sim v_\ell \ \text{en } M_{j+1}
\quad\Longleftrightarrow\quad
v_i \sim v_\ell \ \text{en } M_j,
\]
\[
w_i \sim v_\ell \ \text{en } M_{j+1}
\quad\Longleftrightarrow\quad
v_i \sim v_\ell \ \text{en } M_j,
\]
y además
\[
w \sim w_i \ \text{en } M_{j+1}\qquad\text{para todo } i=0,\dots,n-1.
\]
No se agregan otras aristas.
\end{definition}

\begin{definition}[Familia \(M_j\)]
Se fija \(M_2:=K_2\) y, para todo \(j\ge 2\), se define \(M_{j+1}\) a partir de \(M_j\)
mediante el paso anterior.
\end{definition}

\begin{lemma}\label{lem:M4-avoid-P1}

\begin{theorem}[
\texorpdfstring{\(\chi(G)-1\le \starparam{12}(G)\le \chi(G)\)}
{Bounds for star12}
]
\label{thm:star12-bounds}
Let \(G\) be a graph with \(\chi(G)\ge 2\). Then
\[
\chi(G)-1
\le
\starparam{12}(G)
\le
\chi(G).
\]
\end{theorem}

\begin{proof}
For the lower bound, observe that
\[
\starparam{12}\preceq\bipparam{12},
\]
since both patterns have color set \(\{1,2\}\), while
\(\starparam{12}\) requires only the edge \(12\), whereas
\(\bipparam{12}\) requires the edges \(12\) and \(23\).
Therefore, by Lemma~\ref{lem:pat-monotone},
\[
\bipparam{12}(G)
\le
\starparam{12}(G).
\]
By Theorem~\ref{thm:bip12-chi},
\[
\bipparam{12}(G)=\chi(G)-1.
\]
Hence
\[
\chi(G)-1\le\starparam{12}(G).
\]

For the upper bound, let
\[
k:=\chi(G),
\]
and let
\[
V(G)=C_1\,\dot\cup\,\cdots\,\dot\cup\,C_k
\]
be a proper \(k\)-coloring of \(G\). Take
\[
\mathcal Q:=\{C_1,\dots,C_k\}
\]
and any order \(\prec\) of \(V(G)\).

Let
\[
v_1\prec v_2\prec v_3
\]
be an ordered triple. If the \edgecondition{} does not hold for
\(\starparam{12}\), then
\[
v_1v_2\in E(G).
\]
Since the \(C_i\) are the color classes of a proper coloring, \(v_1\) and
\(v_2\) lie in different parts of \(\mathcal Q\). Hence the
\colorcondition{} holds.

Thus every ordered triple avoids \(\starparam{12}\), so
\[
\starparam{12}(G)\le k=\chi(G).
\]
\end{proof}





\begin{theorem}[
Characterization of the case
\texorpdfstring{\(\starparam{12}(G)=\chi(G)\)}
{star12(G) = chi(G)}
]
\label{thm:star12-equals-chi}
Let \(G\) be a graph, and let
\[
k:=\starparam{12}(G).
\]
The following statements are equivalent:

\begin{enumerate}
\item
\[
k=\chi(G).
\]

\item There exists an optimal solution
\[
S=(\prec,\mathcal Q),
\qquad
\mathcal Q=\{V_1,\dots,V_k\},
\]
that avoids \(\starparam{12}\) in \(G\) and satisfies the following
condition: if \(z\) is the last vertex of the order and
\[
z\in V_h,
\]
then
\[
N_G(z)\cap V_h=\varnothing.
\]
\end{enumerate}
\end{theorem}

\begin{proof}
Suppose first that
\[
k=\chi(G).
\]
Let
\[
V(G)=V_1\,\dot\cup\,\cdots\,\dot\cup\,V_k
\]
be a proper \(k\)-coloring of \(G\), and let \(\prec\) be any total order
of \(V(G)\). Consider the solution
\[
S=(\prec,\mathcal Q),
\qquad
\mathcal Q=\{V_1,\dots,V_k\}.
\]

Since every \(V_i\) is independent, whenever the \edgecondition{} does not
hold for \(\starparam{12}\), the vertices in positions \(1\) and \(2\)
are adjacent and therefore lie in different parts of \(\mathcal Q\).
Hence the \colorcondition{} holds. Thus \(S\) avoids
\(\starparam{12}\) in \(G\).

The solution has value
\[
|\mathcal Q|=k=\starparam{12}(G),
\]
so it is optimal. Let \(z\) be the last vertex of the order, and let
\(V_h\) be the part containing \(z\). Since \(V_h\) is independent,
\[
N_G(z)\cap V_h=\varnothing.
\]

Conversely, suppose that there exists an optimal solution
\[
S=(\prec,\mathcal Q),
\qquad
\mathcal Q=\{V_1,\dots,V_k\},
\]
satisfying the condition in item \(2\). Let \(z\) be the last vertex of
the order, and let \(V_h\) be the part containing \(z\).

We show that every part of \(\mathcal Q\) is independent. Suppose,
towards a contradiction, that there exist adjacent vertices
\[
x,y\in V_i,
\qquad
x\prec y.
\]

If \(y\neq z\), then there exists a vertex \(t\) such that
\[
x\prec y\prec t.
\]
For this ordered triple, the \edgecondition{} does not hold because
\(x\sim y\), while the \colorcondition{} does not hold because
\(x\) and \(y\) lie in the same part \(V_i\). Moreover, the
\nonedgecondition{} never holds for \(\starparam{12}\), since its set of
required non-edges is empty. Therefore, the triple does not avoid
\(\starparam{12}\), a contradiction.

Hence
\[
y=z.
\]
Since \(y\in V_i\) and \(z\in V_h\), we have \(i=h\). Thus
\[
x\in N_G(z)\cap V_h,
\]
contradicting
\[
N_G(z)\cap V_h=\varnothing.
\]

Therefore every \(V_i\) is independent, so \(\mathcal Q\) is a proper
\(k\)-coloring of \(G\). Hence
\[
\chi(G)\le k.
\]
On the other hand, the upper-bound construction in
Theorem~\ref{thm:star12-bounds} gives
\[
k=\starparam{12}(G)\le\chi(G).
\]
Consequently,
\[
k=\chi(G).
\]
\end{proof}




\begin{corollary}[
Characterization of the case
\texorpdfstring{\(\starparam{12}(G)=\chi(G)-1\)}
{star12(G) = chi(G) - 1}
]
\label{cor:star12-equals-chi-minus-one}
Let \(G\) be a graph with
\[
\chi(G)\ge 2.
\]
Then
\[
\starparam{12}(G)=\chi(G)-1
\]
if and only if, for every optimal solution
\[
S=(\prec,\mathcal Q)
\]
avoiding \(\starparam{12}\), the following holds: if \(z\) is the last
vertex of the order and \(V_h\in\mathcal Q\) is the part containing \(z\),
then
\[
N_G(z)\cap V_h\neq\varnothing.
\]
\end{corollary}

\begin{proof}
By Theorem~\ref{thm:star12-equals-chi},
\[
\starparam{12}(G)=\chi(G)
\]
if and only if there exists an optimal solution whose last vertex has no
neighbor in its own part.

Therefore,
\[
\starparam{12}(G)\neq\chi(G)
\]
if and only if, in every optimal solution, the last vertex has a neighbor
in its own part.

Finally, Theorem~\ref{thm:star12-bounds} gives
\[
\chi(G)-1
\le
\starparam{12}(G)
\le
\chi(G).
\]
Hence
\[
\starparam{12}(G)\neq\chi(G)
\quad\Longleftrightarrow\quad
\starparam{12}(G)=\chi(G)-1.
\]

\end{proof}







\paragraph{Intuition.}
In a solution avoiding \(\operatorname{star}_{12}\), an edge whose endpoints lie in the same part can only have its
right endpoint as the last vertex of the order. Indeed, if
\[
x\prec y,\qquad xy\in E(G),
\]
and \(x\) and \(y\) lie in the same part, then any vertex \(z\) with \(y\prec z\) would produce an ordered triple
\[
x\prec y\prec z
\]
that satisfies neither \edgecondition{} nor \colorcondition{} for \(\operatorname{star}_{12}\).
Therefore, to force the endpoints of every edge of a graph \(X\) to lie in different parts, it is enough to force
the last vertex of the order to lie outside \(V(X)\). We achieve this by taking the disjoint union of \(X\) with
a clique \(K_{k+1}\).

\label{thm:star12-np-complete}
For every fixed integer \(k\ge 3\), the restriction of
\(\starparam{12}\)-Avoidance to instances with bound \(k\) is NP-complete.
\end{theorem}

\(k\)-\textsc{Coloring}, and construct
\[

be the corresponding partition into independent sets. Let

whose endpoints lie in the same part.

assume without loss of generality that \(x\prec y\). Since \(z\notin V(X)\) and \(z\) is the last vertex,
\[

\]
If \(x\) and \(y\) were assigned to the same part, then the ordered triple \((x,y,z)\) would satisfy neither
\edgecondition{} nor \colorcondition{} for \(\operatorname{star}_{12}\), a contradiction.

Therefore, the endpoints of every edge of \(X\) lie in different parts of \(\mathcal Q\). The restriction of

\(k\)-colorable.

\begin{theorem}[Characterization of the \(P\)-chromatic number of the star pattern]
\label{thm:star-P-chromatic-number}

\chi_P(G)\le k

\[
|S|\le k
\qquad\text{and}\qquad
\chi(G-S)\le k.

\end{theorem}

\begin{proof}
Suppose first that
\[
\chi_P(G)\le k.
\]
Then there exists a \(P\)-coloring
\[
V(G)=V_1\,\dot\cup\,\cdots\,\dot\cup\,V_q,
\qquad
q\le k.
\]

\tau(G[V_i])\le1

\qquad

\]
Then
\[
|S|\le q\le k.
\]

Moreover, each \(V_i\setminus S_i\) is independent. Hence these sets form
a proper coloring of \(G-S\) with at most \(q\le k\) colors. Therefore,
\[
\chi(G-S)\le k.

\[
S\subseteq V(G)

|S|\le k

|S_i|\le1
\qquad

\[
\chi_P(G)\le k.

\begin{corollary}[
Polynomial-time solvability of
\texorpdfstring{\(\starparam{13}\)-Avoidance with bound \(2\)}
{star13-Avoidance with bound 2}
]
\label{cor:star13-k2}
The \(\starparam{13}\)-Avoidance problem with bound \(2\) can be solved
in polynomial time.
\end{corollary}

\begin{proof}
By Observation~\ref{obs:star13-avoidance-characterization},
\[
\starparam{13}(G)\le2
\]
if and only if there exists a set \(S\subseteq V(G)\) such that
\[
|S|\le2
\qquad\text{and}\qquad
\chi(G-S)\le2.
\]
Equivalently, \(G-S\) is bipartite.

Thus, it is enough to enumerate all
\[
1+|V(G)|+\binom{|V(G)|}{2}
\]
sets \(S\subseteq V(G)\) of size at most two and test whether \(G-S\)
is bipartite. Hence the problem can be solved in polynomial time.
\end{proof}


\paragraph{Intuition.}
By the equality
\[
\starparam{13}(G)=\chi_{\starparam{13}}(G),
\]
we may view \(\starparam{13}\) as a partition parameter. In each part, the
induced graph must belong to the class associated with the star pattern,
that is, it must have a vertex cover of size at most one.

To encode a proper \(k\)-coloring, we add a clique \(K_{2k}\). Since a part
whose induced graph has vertex cover number at most one cannot contain three
vertices of this clique, any partition into at most \(k\) parts is forced to
place exactly two clique vertices in each part. These two vertices already
form an edge, which prevents the same part from containing an edge of the
original graph.




\begin{theorem}[NP-completeness of \(\starparam{13}\)-Avoidance]
\label{thm:star13-np-complete}
For every fixed integer \(k\ge 3\), the restriction of
\(\starparam{13}\)-Avoidance to instances with bound \(k\) is NP-complete.
\end{theorem}

\begin{proof}
We prove NP-hardness by a reduction from
\(k\)-\textsc{Coloring}.

Let \(X\) be an instance of \(k\)-\textsc{Coloring}. We construct
\[
G:=X\,\dot\cup\,K_{2k},
\]
that is, the disjoint union of \(X\) and a clique on \(2k\) vertices.

We claim that
\[
X\text{ is \(k\)-colorable}
\quad\Longleftrightarrow\quad
\starparam{13}(G)\le k.
\]

\medskip
\noindent\textbf{(\(\Rightarrow\))}
Suppose that \(X\) is \(k\)-colorable, and let
\[
V(X)=I_1\,\dot\cup\,\cdots\,\dot\cup\,I_k
\]
be a proper \(k\)-coloring, where each \(I_i\) is independent.

Partition the vertices of \(K_{2k}\) into \(k\) pairs
\[
P_1,\dots,P_k,
\]
and define
\[
V_i:=I_i\cup P_i
\qquad
\text{for }i=1,\dots,k.
\]

Since \(G\) is a disjoint union, there are no edges between \(I_i\) and
\(P_i\). Moreover, \(I_i\) is independent and \(P_i\) induces a single
edge. Hence
\[
\tau(G[V_i])\le1
\qquad
\text{for every }i.
\]
Equivalently,
\[
G[V_i]\in\mathcal C_{\starparam{13}}
\qquad
\text{for every }i.
\]
Thus
\[
\chi_{\starparam{13}}(G)\le k.
\]
By Corollary~\ref{cor:13-colored-partition-equality},
\[
\starparam{13}(G)=\chi_{\starparam{13}}(G),
\]
and therefore
\[
\starparam{13}(G)\le k.
\]

\medskip
\noindent\textbf{(\(\Leftarrow\))}
Suppose now that
\[
\starparam{13}(G)\le k.
\]
Again using Corollary~\ref{cor:13-colored-partition-equality}, there exists
a partition
\[
V(G)=V_1\,\dot\cup\,\cdots\,\dot\cup\,V_q,
\qquad
q\le k,
\]
such that
\[
\tau(G[V_i])\le1
\qquad
\text{for every }i=1,\dots,q.
\]

Consider the copy of \(K_{2k}\). No part \(V_i\) can contain three of its
vertices, since three vertices of a clique induce a \(K_3\), and
\[
\tau(K_3)=2.
\]
Therefore,
\[
|V_i\cap V(K_{2k})|\le2
\qquad
\text{for every }i.
\]

Since the clique contains \(2k\) vertices and there are at most \(k\) parts,
it follows that necessarily
\[
q=k
\]
and every part contains exactly two vertices of \(K_{2k}\). In particular,
each \(G[V_i]\) contains an edge whose two endpoints belong to \(K_{2k}\).

We show that
\[
V_i\cap V(X)
\]
is independent for every \(i\). Suppose, towards a contradiction, that
there exist
\[
x,y\in V_i\cap V(X)
\]
such that \(xy\in E(X)\).

The part \(V_i\) already contains an edge whose endpoints lie in
\(K_{2k}\). Since \(G\) is the disjoint union of \(X\) and \(K_{2k}\),
this edge is disjoint from \(xy\). Hence \(G[V_i]\) contains two disjoint
edges. Any vertex cover of \(G[V_i]\) must contain at least one endpoint
of each of these edges, and therefore
\[
\tau(G[V_i])\ge2,
\]
contradicting the choice of the partition.

Thus every set \(V_i\cap V(X)\) is independent. Consequently,
\[
V(X)
=
\bigl(V_1\cap V(X)\bigr)
\,\dot\cup\,\cdots\,\dot\cup\,
\bigl(V_k\cap V(X)\bigr)
\]
is a proper \(k\)-coloring of \(X\). Hence \(X\) is \(k\)-colorable.

The construction of \(G\) is polynomial, so
\(\starparam{13}\)-Avoidance with bound \(k\) is NP-hard. By
Observation~\ref{obs:avoidance-in-NP}, it is NP-complete.
\end{proof}

\begin{definition}[\(\mathcal P\)-removable vertex]

whose last vertex is \(v\).

\begin{theorem}\label{thm:forrt-given-order-np-complete}
The \(\forrt\)-Avoidance problem with a given order is NP-complete.
More precisely, for every fixed \(k\ge 3\), deciding whether there exists a partition into at most \(k\) parts
that avoids \(\forrt\) with respect to a given order is NP-complete.
\end{theorem}

\begin{proof}[Proof sketch]
We reduce from \(k\)-\textsc{Coloring}. Let \(X\) be an instance with
\[
V(X)=\{x_1,\dots,x_n\},
\]
ordered arbitrarily as
\[
x_1\prec x_2\prec\cdots\prec x_n.
\]
For each edge \(x_i x_j\in E(X)\), with \(i<j\), we create a witness vertex \(w_{ij}\). We keep the original
edge \(x_i x_j\), and we make \(w_{ij}\) adjacent only to \(x_j\). Thus,
\[
V(G)=V(X)\cup \{w_{ij}: x_i x_j\in E(X),\ i<j\},
\]
and
\[
E(G)=E(X)\cup \{w_{ij}x_j : x_i x_j\in E(X),\ i<j\}.
\]

For each \(j\), let
\[
W_j:=\{w_{ij}: i<j \text{ and } x_i x_j\in E(X)\}.
\]
The given order is
\[
x_1\prec W_2\prec x_2\prec W_3\prec x_3\prec\cdots\prec W_n\prec x_n,
\]
where the vertices inside each \(W_j\) are ordered arbitrarily. Hence, for every edge \(x_i x_j\in E(X)\), with \(i<j\),
we have
\[
x_i\prec w_{ij}\prec x_j,
\qquad
x_i\sim x_j,
\qquad
w_{ij}\sim x_j.
\]

If \(X\) has a proper \(k\)-coloring, keep the colors on the original vertices and assign each witness \(w_{ij}\) a color
different from the color of \(x_j\) (for instance, the color of \(x_i\)). This gives a partition that avoids \(\forrt\).

Conversely, if a \(\forrt\)-avoiding partition exists, then for every edge \(x_i x_j\in E(X)\), with \(i<j\), the triple
\[
x_i\prec w_{ij}\prec x_j
\]
forces \(x_i\) and \(x_j\) to lie in different parts. Therefore, the restriction of the partition to \(V(X)\) is a proper
\(k\)-coloring of \(X\).
\end{proof}

\begin{theorem}\label{thm:forst-general-np-complete}
The \(\forst\)-Avoidance problem is NP-complete.
More precisely, for every fixed \(k\ge 3\), deciding whether a graph admits a
\(\forst\)-avoiding solution with at most \(k\) parts is NP-complete.
\end{theorem}

\begin{proof}[Proof sketch]
We reduce from \(k\)-\textsc{Coloring}. Given an instance \(X\), for every vertex
\(x\in V(X)\) we create an independent block
\[
B_x:=\{x_1,x_2\}.
\]
For every edge \(xy\in E(X)\), we join \(B_x\) and \(B_y\) completely, and we add no other edges. Thus
\[
V(G)=\bigcup_{x\in V(X)} B_x,
\]
and
\[
E(G)=\{x_i y_j:\ xy\in E(X),\ i,j\in\{1,2\}\}.
\]

If \(X\) has a proper \(k\)-coloring, put both vertices of \(B_x\) in the color class of \(x\). Since adjacent
vertices of \(X\) receive different colors, no edge of \(G\) lies inside a part, and hence the solution avoids
\(\forst\) for any order.

Conversely, suppose that \(G\) has a \(\forst\)-avoiding solution
\[
S=(\prec,\mathcal Q),
\qquad |\mathcal Q|\le k.
\]
For each \(x\in V(X)\), let \(\ell_x\) be the last vertex of the block \(B_x=\{x_1,x_2\}\) in the order \(\prec\).
We color \(x\) with the part of \(\mathcal Q\) that contains \(\ell_x\).

We claim that this gives a proper \(k\)-coloring of \(X\). Let \(xy\in E(X)\). We need to show that
\(\ell_x\) and \(\ell_y\) lie in different parts of \(\mathcal Q\). Suppose, to the contrary, that they lie in the
same part. Without loss of generality, assume
\[
\ell_x \prec \ell_y.
\]
Let \(x'\) be the other vertex of \(B_x\). Since \(\ell_x\) is the last vertex of \(B_x\), we have
\[
x'\prec \ell_x\prec \ell_y.
\]
Moreover, because \(xy\in E(X)\), the construction makes \(B_x\) complete to \(B_y\). Hence
\[
x'\sim \ell_y
\qquad\text{and}\qquad
\ell_x\sim \ell_y.
\]
Finally, \(\ell_x\) and \(\ell_y\) lie in the same part by assumption. Therefore the ordered triple
\[
x'\prec \ell_x\prec \ell_y
\]
realizes the forbidden pattern \(\forst\), contradicting that \(S\) avoids \(\forst\).

Thus, for every edge \(xy\in E(X)\), the vertices \(x\) and \(y\) receive different colors. Hence \(X\) is
\(k\)-colorable.
\end{proof}

\[

where \(A\) is a set of elements, \(B\) is a set of precedence constraints \((u,v)\), requiring
\(u\prec v\), and \(T\) is a family of disjoint triples \((a,b,c)\), requiring that the order
\(a\prec b\prec c\) does \emph{not} occur. In the notation of Guttmann--Maucher, this corresponds to family \(4\) with
\[
P=S_3\setminus\{(123)\}.

First, we subdivide the precedence constraints. For every \((u,v)\in B\), add a new element \(z_{uv}\) and replace

\[
u\prec z_{uv},
\qquad
z_{uv}\prec v.
\]
Let \((A',B',T)\) be the resulting instance. This transformation preserves satisfiability.

We now construct a graph \(G\) and a fixed partition \(\mathcal Q\). The base vertices of \(G\) are the elements of
\(A'\). For every pair \((u,v)\in B'\), add two auxiliary vertices \(p_{uv}\) and \(q_{uv}\). Thus
\[
V(G)=A'\cup \{p_{uv},q_{uv}:(u,v)\in B'\}.
\]

The partition \(\mathcal Q\) is defined as follows. For every triple \((a,b,c)\in T\), put \(a\) and \(c\) in the
same part, and put \(b\) in a different part. Since the triples are disjoint, this creates no conflicts. Every base
vertex not appearing in a triple is put in its own part. Finally, for each \((u,v)\in B'\), put \(p_{uv}\) and
\(q_{uv}\) in the same part as \(v\).

The edges of \(G\) are the following. For every triple \((a,b,c)\in T\), add
\[
ac,\qquad bc,
\]
and do not add the edge \(ab\). For every pair \((u,v)\in B'\), add
\[
up_{uv},\quad vp_{uv},\quad uq_{uv},\quad vq_{uv}.
\]
No other edges are added.

The triple gadget forbids the order \(a\prec b\prec c\): if this order occurred, then the triple
\(a\prec b\prec c\) would have the required edges \(ac\) and \(bc\), and \(a,c\) would lie in the same part, giving an
occurrence of \(\forrt\).

The pair gadget forces \(u\prec v\). Indeed, suppose that a valid order had \(v\prec u\). If one of the auxiliaries
appears after \(u\), say
\[
v\prec u\prec p_{uv},
\]
then this triple realizes \(\forrt\), since \(v\sim p_{uv}\), \(u\sim p_{uv}\), and \(v,p_{uv}\) lie in the same part.
If both auxiliaries appear before \(u\), then two of them are ordered, say
\[
p_{uv}\prec q_{uv}\prec u,
\]
and this triple realizes \(\forrs\), since \(p_{uv},q_{uv}\) lie in the same part and both are adjacent to \(u\).
Thus every valid order must satisfy \(u\prec v\).

Conversely, assume that the ordering instance \((A',B',T)\) is satisfiable. Extend a satisfying order to the auxiliary
vertices by placing, for every \((u,v)\in B'\),
\[
u\prec \cdots \prec v \prec p_{uv}\prec q_{uv}.
\]
The subdivision step ensures that the endpoints \(u\) and \(v\) of every pair in \(B'\) lie in different parts of
\(\mathcal Q\). With this extension, the pair gadgets do not create occurrences of either \(\forrs\) or \(\forrt\),
and the triple gadgets are valid because the order \(a\prec b\prec c\) is avoided for every \((a,b,c)\in T\).

Therefore, the original ordering instance is satisfiable if and only if the constructed instance \((G,\mathcal Q)\)
admits an order avoiding both \(\forrs\) and \(\forrt\). The construction is polynomial. The problem is in NP because
a certificate is an order \(\prec\), and checking whether \((\prec,\mathcal Q)\) avoids both patterns can be done in
polynomial time. Hence the problem is NP-complete.

\begin{theorem}\label{thm:forrs-st-given-partition-np-complete}
The \(\mathcal F_{rs\text{-}st}\)-Avoidance problem with a given partition is NP-complete.
\end{theorem}

\begin{proof}[Proof sketch]
We reduce from the following NP-complete variant of \textsc{Betweenness with precedence constraints}:
an instance is \((A,B,T)\), where \(A\) is a set of elements, \(B\) is a set of precedence constraints
\((u,v)\) requiring \(u\prec v\), and \(T\) is a set of disjoint triples \((a,b,c)\) requiring \(b\) to lie
between \(a\) and \(c\), that is, either \(a\prec b\prec c\) or \(c\prec b\prec a\).

First, for each \((u,v)\in B\), add a new element \(z_{uv}\) and replace \(u\prec v\) by
\[
u\prec z_{uv},
\qquad
z_{uv}\prec v.
\]
Let \((A',B',T)\) be the resulting instance.

We construct a graph \(G\) and a fixed partition \(\mathcal Q\).

The base vertices of \(G\) are the elements of \(A'\). For every pair \((u,v)\in B'\), add two auxiliary vertices
\(p_{uv}\) and \(q_{uv}\). Thus
\[
V(G)=A'\cup \{p_{uv},q_{uv}:(u,v)\in B'\}.
\]

The partition \(\mathcal Q\) is defined as follows. For every triple \((a,b,c)\in T\), put \(a\) and \(c\) in the
same part, and put \(b\) in a different part. Since the triples are disjoint, this creates no conflicts.
Every base vertex not appearing in a triple is put in its own part. Finally, for each \((u,v)\in B'\), put
\(p_{uv}\) and \(q_{uv}\) in the same part as \(u\).

The edges of \(G\) are the following. For every triple \((a,b,c)\in T\), add the triangle
\[
ab,\quad bc,\quad ac.
\]
For every pair \((u,v)\in B'\), add
\[
up_{uv},\quad vp_{uv},\quad uq_{uv},\quad vq_{uv}.
\]
No other edges are added.

The triple gadgets force betweenness: since \(a\) and \(c\) are in the same part and \(a,b,c\) form a triangle,
placing \(b\) after both \(a,c\) creates an occurrence of \(\forrs\), while placing \(b\) before both \(a,c\)
creates an occurrence of \(\forst\). Hence \(b\) must lie between \(a\) and \(c\).

The pair gadgets force precedence. For \((u,v)\in B'\), the vertices \(p_{uv},q_{uv}\) are in the part of \(u\) and
are common neighbors of \(u\) and \(v\). If an order had \(v\prec u\), then either one auxiliary appears after \(u\),
creating an occurrence of \(\forst\), or both auxiliaries appear before \(u\), in which case the two auxiliaries together
with \(u\) create an occurrence of \(\forrs\). Therefore every valid order must satisfy \(u\prec v\).

Conversely, any order satisfying \((A',B',T)\) can be extended to the auxiliary vertices by placing, for each
\((u,v)\in B'\),
\[
u\prec \cdots \prec v \prec p_{uv}\prec q_{uv}.
\]
The subdivision step ensures that \(u\) and \(v\) are in different parts of \(\mathcal Q\), so the pair gadgets do not
create occurrences of either \(\forrs\) or \(\forst\). The triple gadgets are valid because each \(b\) lies between
the corresponding \(a\) and \(c\).

Thus the original instance is satisfiable if and only if the constructed instance \((G,\mathcal Q)\) admits an order
avoiding both \(\forrs\) and \(\forst\). The construction is polynomial, and the problem is in NP because an order can
be checked in polynomial time. Hence the problem is NP-complete.
\end{proof}

\begin{theorem}\label{thm:forrt-st-given-order-np-complete}
The \(\mathcal F_{rt\text{-}st}\)-Avoidance problem with a given order is NP-complete.
More precisely, for every fixed \(k\ge 3\), deciding whether there exists a partition into at most \(k\) parts
that avoids both \(\forrt\) and \(\forst\) with respect to a given order is NP-complete.
\end{theorem}

\begin{proof}[Proof sketch]
We reduce from \(k\)-\textsc{Coloring}. Let \(X\) be an instance with
\[
V(X)=\{x_1,\dots,x_n\},
\]
ordered arbitrarily as
\[
x_1\prec x_2\prec\cdots\prec x_n.
\]
For each edge \(x_i x_j\in E(X)\), with \(i<j\), add one witness vertex \(w_{ij}\), adjacent only to \(x_j\).
We keep the original edge \(x_i x_j\). Thus,
\[
V(G)=V(X)\cup \{w_{ij}:x_i x_j\in E(X),\ i<j\},
\]
and
\[
E(G)=E(X)\cup \{w_{ij}x_j:x_i x_j\in E(X),\ i<j\}.
\]
The given order puts all witnesses first, in arbitrary order, and then the original vertices:
\[
\text{all } w_{ij}\prec x_1\prec x_2\prec\cdots\prec x_n.
\]
In particular, for every edge \(x_i x_j\in E(X)\), with \(i<j\), we have
\[
w_{ij}\prec x_i\prec x_j,
\qquad
w_{ij}\sim x_j,
\qquad
x_i\sim x_j.
\]

If \(X\) has a proper \(k\)-coloring, keep the colors on the original vertices and assign each witness
\(w_{ij}\) the color of \(x_i\). Then every earlier neighbor of an original vertex has color different from it,
so no ordered triple realizes either \(\forrt\) or \(\forst\).

Conversely, suppose that a partition avoiding both \(\forrt\) and \(\forst\) exists. For each edge
\(x_i x_j\in E(X)\), with \(i<j\), the triple
\[
w_{ij}\prec x_i\prec x_j
\]
has the required edges to \(x_j\). Since \(\forst\) is avoided, \(x_i\) and \(x_j\) must lie in different parts.
Therefore, the restriction of the partition to \(V(X)\) is a proper \(k\)-coloring of \(X\).

Thus \(X\) is \(k\)-colorable if and only if the constructed ordered graph admits a partition into at most \(k\)
parts avoiding \(\mathcal F_{rt\text{-}st}\). The problem is in NP, so it is NP-complete.
\end{proof}

\begin{theorem}\label{thm:forrt-st-general-np-complete}
The \(\mathcal F_{rt\text{-}st}\)-Avoidance problem is NP-complete.
More precisely, for every fixed \(k\ge 3\), deciding whether a graph admits a solution with at most \(k\) parts
that avoids both \(\forrt\) and \(\forst\) is NP-complete.
\end{theorem}

\begin{proof}[Proof sketch]
We reduce from \(k\)-\textsc{Coloring}. Given an instance \(X\), we construct \(G\) by adding one new vertex \(F\)
adjacent to every vertex of \(X\), while keeping all edges of \(X\). That is,
\[
V(G)=V(X)\cup\{F\},
\qquad
E(G)=E(X)\cup \{Fx:x\in V(X)\}.
\]

If \(X\) has a proper \(k\)-coloring, choose one color \(\alpha\), assign color \(\alpha\) to \(F\), and order the vertices as
\[
F \prec \text{vertices of color }\alpha \prec \text{all remaining vertices}.
\]
Then no vertex has two earlier neighbors one of which lies in its own part, so the solution avoids both
\(\forrt\) and \(\forst\).

Conversely, suppose that \(G\) has a solution avoiding both \(\forrt\) and \(\forst\), with at most \(k\) parts.
Let \(\alpha\) be the part containing \(F\), and write
\[
A:=\{x\in V(X):x\prec F\},
\qquad
B:=\{x\in V(X):F\prec x\}.
\]
The key observation is the following: if a vertex \(t\) has at least two earlier neighbors, then all its earlier
neighbors must lie in parts different from the part of \(t\).

From this observation, the coloring induced by the solution is already proper on \(B\), and also on the edges between
\(A\) and \(B\). The only possible monochromatic edges of \(X\) lie inside \(A\). Moreover, for each part \(\beta\),
the subgraph induced by the vertices of \(A\) in part \(\beta\) is a forest: if it contained a cycle, the last vertex
of the cycle in the order would have two earlier neighbors in its own part, contradicting the observation.

We now recolor the monochromatic forest components inside \(A\). For each part \(\beta\neq \alpha\), root every tree
of the forest induced by \(A\cap \beta\), keep one side of the bipartition in color \(\beta\), and recolor the other
side with the auxiliary color \(\alpha\). The vertices recolored with \(\alpha\) form an independent set, again by the
observation above. Thus this recoloring destroys all monochromatic edges inside \(A\), and creates no new monochromatic
edges.

Hence we obtain a proper \(k\)-coloring of \(X\). Therefore \(X\) is \(k\)-colorable if and only if \(G\) admits a
solution with at most \(k\) parts avoiding \(\mathcal F_{rt\text{-}st}\). The construction is polynomial, and the
problem is in NP, so it is NP-complete.
\end{proof}

\begin{definition}[First-two-symmetric family]
Let \(\mathcal P\) be a family of colored patterns. We say that \(\mathcal P\) is
\emph{first-two-symmetric} if, for every graph \(H\), every partition \(\mathcal Q\) of \(V(H)\),
and every three distinct vertices \(x,y,z\in V(H)\), the ordered triple
\[
(x,y,z)
\]
avoids \(\mathcal P\) in \(H\) if and only if the ordered triple
\[
(y,x,z)
\]
avoids \(\mathcal P\) in \(H\).

In other words, once the last vertex of the ordered triple is fixed, the order of the first two vertices does not matter.
\end{definition}




\begin{algorithm}[ht]
\caption{\textsc{Given-Partition-\(\mathcal P\)}}
\label{alg:generic-given-partition}
\begin{algorithmic}[1]
\Require A graph \(G=(V(G),E(G))\), a partition \(\mathcal Q=\{V_1,\dots,V_k\}\) of \(V(G)\), and a family of colored patterns \(\mathcal P\).
\Ensure \textbf{YES} and an order \(\prec\) if such an order exists; otherwise \textbf{NO}.

\State \(A \gets V(G)\)
\State \(L \gets [\,]\) \Comment{\(L\) stores the vertices in deletion order}

\While{\(A\neq\varnothing\)}
    \If{there exists a \(\mathcal P\)-removable vertex \(v\in A\) in \(G[A]\) with respect to \(\mathcal Q\)}
        \State append \(v\) to \(L\)
        \State \(A\gets A\setminus\{v\}\)
    \Else
        \State \Return \textbf{NO}
    \EndIf
\EndWhile

\State Let \(\prec\) be the order obtained by reading \(L\) from right to left
\State \Return \((\textbf{YES},\prec)\)
\end{algorithmic}
\end{algorithm}


\begin{lemma}[Correctness of Algorithm~\ref{alg:generic-given-partition}]
\label{lem:generic-given-partition-correctness}
Let \(\mathcal P\) be a first-two symmetric family of colored patterns and let \(G\) be a graph.
Then Algorithm~\ref{alg:generic-given-partition} returns \textbf{YES} if and only if there exists an order
\(\prec\) of \(V(G)\) such that
\[
(\prec,\mathcal Q)
\]
avoids every pattern in \(\mathcal P\) in \(G\).
\end{lemma}


\begin{proof}
First suppose that the algorithm returns \textbf{YES}. Let \(\prec\) be the order obtained by reading the deletion
list \(L\) from right to left. We show that \((\prec,\mathcal Q)\) avoids \(\mathcal P\) in \(G\).

Consider any ordered triple
\[
x\prec y\prec z
\]
in the final order. When \(z\) was deleted by the algorithm, both \(x\) and \(y\) were still active, because they
appear before \(z\) in the final order. Since \(z\) was \(\mathcal P\)-removable in the current induced subgraph,
the triple \((x,y,z)\) avoids every pattern in \(\mathcal P\). Therefore every ordered triple avoids \(\mathcal P\),
and so \((\prec,\mathcal Q)\) is a valid solution.

Conversely, suppose that there exists an order \(\prec\) of \(V(G)\) such that \((\prec,\mathcal Q)\) avoids
\(\mathcal P\). We prove that the algorithm returns \textbf{YES}.

At any step, let \(H=G[A]\) be the current induced subgraph. The restriction of \(\prec\) to \(A\) still avoids
\(\mathcal P\), because deleting vertices cannot create a forbidden ordered triple. Let \(v\) be the last vertex of
this restricted order.

We claim that \(v\) is \(\mathcal P\)-removable in \(H\). Take any two distinct vertices
\(x,y\in V(H)\setminus\{v\}\). In the restricted order, either
\[
x\prec y\prec v
\]
or
\[
y\prec x\prec v.
\]
The corresponding ordered triple avoids \(\mathcal P\), because the restricted order is valid. Since \(\mathcal P\)
is first-two symmetric, the other ordering of the first two vertices also avoids \(\mathcal P\). Hence \((x,y,v)\)
avoids \(\mathcal P\) for every pair \(x,y\), so \(v\) is \(\mathcal P\)-removable.

Thus, whenever a valid order exists for the current induced subgraph, a \(\mathcal P\)-removable vertex exists.
Therefore the algorithm cannot return \textbf{NO}.
\end{proof}

|N_H(v)\cap V_i|\le 1
\qquad\text{for every } i=1,\dots,k.
\]
We maintain counters
\[
c(v,i):=|N_H(v)\cap V_i|
\]
for every active vertex \(v\) and every part \(V_i\), together with the number of parts \(V_i\) for which
\(c(v,i)\ge 2\). Then \(v\) is removable exactly when this number is zero.

The counters can be initialized in time \(O(nk+m)\). When a vertex \(u\in V_j\) is deleted, only the counters
\(c(v,j)\) for active neighbors \(v\) of \(u\) change. Thus each edge is processed at most once during the deletion
process. Using a queue of currently removable vertices, the total running time is
\[
O(nk+m).
\]

\medskip
\noindent\textbf{The family \(\mathcal F_{13\text{-}23}\).}
By Lemma~\ref{lem:removable-for13-23}, a vertex \(v\) is \(\mathcal F_{13\text{-}23}\)-removable in the current
induced subgraph \(H\) if and only if
\[
\deg_H(v)\le 1
\qquad\text{or}\qquad
N_H(v)\cap \mathcal Q(v)=\varnothing,
\]
where \(\mathcal Q(v)\) denotes the part of \(\mathcal Q\) containing \(v\).

We maintain the current degree \(\deg_H(v)\) of each active vertex \(v\), and also the number
\[
s(v):=|N_H(v)\cap \mathcal Q(v)|
\]
of active neighbors of \(v\) in its own part. When a vertex is deleted, these quantities are updated only for its
active neighbors. Hence each edge is processed at most once, and the total running time is
\[
O(n+m).
\]

\medskip
\noindent\textbf{The family \(\mathcal F_{12\text{-}23\text{-}13}\).}
By Lemma~\ref{lem:removable-for12-23-13}, a vertex \(v\) is \(\mathcal F_{12\text{-}23\text{-}13}\)-removable in
the current induced subgraph \(H\) if and only if either
\[
\deg_H(v)\le 1,
\]
or both
\[
N_H(v)\cap \mathcal Q(v)=\varnothing
\]
and
\[
|N_H(v)\cap V_i|\le 1
\qquad\text{for every } i=1,\dots,k.
\]
Thus we combine the counters used in the previous two cases: the degree \(\deg_H(v)\), the same-part counter
\(s(v)=|N_H(v)\cap \mathcal Q(v)|\), and the counters
\[
c(v,i):=|N_H(v)\cap V_i|.
\]
Again, these can be initialized in time \(O(nk+m)\), and each edge is processed at most once during the deletion
process. Hence the total running time is
\[
O(nk+m).
\]

Therefore the given-partition avoidance problem is polynomial-time solvable for all three families, with the stated
running times.
\end{proof}

By construction, the vertices \(x\), \(y\), and \(w^a_{xy}\) are assigned to pairwise different parts. Therefore the
color conditions \(C=\{1,2\}\), \(C=\{1,3\}\), and \(C=\{2,3\}\) all fail. Hence the triple avoids
\(\operatorname{for}_{12}\), \(\operatorname{for}_{13}\), and \(\operatorname{for}_{23}\).

Thus the constructed solution avoids \(\mathcal F_{12\text{-}23\text{-}13}\).

The converse follows from Lemma~\ref{lem:general-12-common-converse} with \(D=\{12,23,13\}\).
\end{proof}
